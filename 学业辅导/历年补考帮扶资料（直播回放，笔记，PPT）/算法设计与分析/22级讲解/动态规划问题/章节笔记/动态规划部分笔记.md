<h3>区间dp的模板</h3>
<pre><code>for(int len=2;len&lt;=n;len++)  //区间长度，区间长度为1时用初始化
for(int i=1;i&lt;=n;i++)        //枚举起点  
{  
int j=i+len-1;           //区间终点  
if(j&gt;n) break;           //越界结束  
for(int k=i;k&lt;j;k++)     //枚举分割点，构造状态转移方程  
{  
    dp[i][j]=max(dp[i][j],dp[i][k]+dp[k+1][j]+w[i][j]);  //加的值会根据具体情况改变
}  
}  
</code></pre>
<h3>0-1背包模板</h3>
<pre><code>for (int i = 0; i &lt; n; i++) {
	for (int v = V; v &gt;= w[i]; v--) {//逆向枚举
		//状态转移方程
		dp[v] = max(dp[v], dp[v - w[i]] + c[i]);
	}
}
</code></pre>
<h3>完全背包问题</h3>
<pre><code>for (int i = 0; i &lt; n; i++) {
	for (int v = w[i]; v &lt;= V; v++) {//正向枚举
		//状态转移方程
		dp[v] = max(dp[v], dp[v - w[i]] + c[i]);
	}
}
</code></pre>
